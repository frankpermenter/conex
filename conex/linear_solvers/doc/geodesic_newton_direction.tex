\section{Problem and notation}

We solve the conic program
\[
  \min_{x}\; c^T x \quad \text{s.t.}\quad Ax + b \succeq 0,
\]
where $A$ is the constraint operator, $b$ the affine term, $c$ the
cost, and $\succeq 0$ denotes membership in a symmetric cone with
identity element~$e$ and quadratic representation $P(X)v$.


\subsection{Perturbed system ($\theta$-centering)}

We consider a $\tau$-scaled and $\theta$-perturbed
central path conditions:
\begin{align}
  A^T \lambda &= \tau\,c + \theta\,(A^T e-c), \label{eq:dual-theta} \\
  s &= Ax + \tau b + \theta(e-b)  , \label{eq:primal-theta} 
\end{align}

\subsection{Geodesic parameterization}
We use the Geodesic Newton direction $d$ defined via
\[
  s = \frac{1}{k}\,P(w^{-1/2})(e - d), \qquad
  \lambda = \frac{1}{k}\,P(w^{1/2})(e + d),
\]
where $k = 1/\sqrt{\mu}$ and $d$ is the Newton direction in $v$-space.
The Gram matrix is $G = A^T P(w) A$.

\section{Three-term decomposition}

The Newton step at the current scaling $W$ is decomposed into three
components that are independent of any blending parameters.

\subsection{Gram solves}

\[
  G\,y_0 = A^T(2w),
\]
\[
  G\,y_1^{(0)} = -\bigl(c + A^T P(w)\,b\bigr),
\]
\[
  G\,y_1^{(\theta)} = c + A^T P(w)\,b - A^T\bigl(e + P(w)\,e\bigr).
\]

\subsection{Direction components}

\[
  d_0 = e - \PW(A\,y_0),
\]
\[
  d_1^{(0)} = -\PW(b + A\,y_1^{(0)}),
\]
\[
  d_1^{(\theta)} = \PW(b - e - A\,y_1^{(\theta)}).
\]

All six quantities ($y_0, y_1^{(0)}, y_1^{(\theta)}, d_0, d_1^{(0)},
d_1^{(\theta)}$) depend only on $W$, the original data $(b, c)$, and
the constraint operator $A$.  They are computed with one factorization
and three back-solves.

\section{Parameterized direction}

Given independent parameters $\tau$ (original-problem weight) and
$\theta$ (identity-centering weight), the Newton direction is
\[
  \boxed{d(k,\tau,\theta) = d_0 + k\bigl(\tau\,d_1^{(0)}
    + \theta\,d_1^{(\theta)}\bigr),}
\]
which is affine in $k$, $\tau$, and $\theta$.

The corresponding variable is
\[
  y(k,\tau,\theta) = y_0 + k\bigl(\tau\,y_1^{(0)}
    + \theta\,y_1^{(\theta)}\bigr).
\]

\subsection{Special cases}

\begin{itemize}
  \item $\tau = 1,\;\theta = 0$: original problem,
        $d = d_0 + k\,d_1^{(0)}$.
  \item $\tau = 0,\;\theta = 1$: feasibility/centering problem with
        $b = e$, $c = A^T e$, \; $d = d_0 + k\,d_1^{(\theta)}$.
  \item $\tau = 1-\theta$: single-parameter blend
        $b_\theta = \theta\,e + (1-\theta)\,b$, \;
        $c_\theta = \theta\,A^T e + (1-\theta)\,c$.
\end{itemize}

\section{Minimum-norm $k$ at fixed $(\tau,\theta)$}

Define $\tilde{d}_1 = \tau\,d_1^{(0)} + \theta\,d_1^{(\theta)}$.
The squared norm of the direction is
\[
  \|d(k)\|^2 = \|d_0\|^2 + 2k\,\langle d_0,\,\tilde{d}_1\rangle
               + k^2\,\|\tilde{d}_1\|^2,
\]
which is minimized at
\[
  \boxed{k^*(\tau,\theta)
    = -\frac{\langle d_0,\, \tau\,d_1^{(0)}
             + \theta\,d_1^{(\theta)}\rangle}
            {\|\tau\,d_1^{(0)}
             + \theta\,d_1^{(\theta)}\|^2}.}
\]


\section{Duality identity}

\subsection{General form}

For the QP with equality constraints
\[
  \min_x\; c^T x + \tfrac{1}{2} x^T Q x
  \quad \text{s.t.}\quad Ax + b \succeq 0,\quad Cx = d,
\]
the KKT stationarity condition is $c + Qx = A^T\lambda + C^T\nu$.
Multiplying by~$x$:
\[
  c^T x + x^T Q x = \lambda^T(Ax) + \nu^T(Cx)
  = \langle s - b,\,\lambda\rangle + d^T\nu
  = \langle s,\lambda\rangle - b^T\lambda + d^T\nu.
\]
In the HSD embedding with lifted variables ($x_{\mathrm{phys}} = x/\tau$),
Combined with $\kappa = \mu/\tau$ and the central-path target
$\langle s,\lambda\rangle + \kappa = \theta\,R$, the identity becomes
\[
  \boxed{b^T \lambda + c^T x + d^T \nu
    + \frac{x^T Q x}{\tau} + \frac{\mu}{\tau}
    = \theta\,\bigl(\langle b, e\rangle + 1\bigr).}
\]
At the trivially centered point ($\theta = 1$, $\tau = 1$, $k = 1$,
$W = e$, $d = 0$): $\lambda = e$, $x = 0$, $\mu = 1$, and
\[
  b^T e + 0 + 0 + 0 + 1 = \langle b, e\rangle + 1. \qquad\checkmark
\]

\subsection{Decomposition of the quadratic term}

The primal variable $x(k, \tau, \theta) = y_0/k + \tau\,y_1^{(0)}
+ \theta\,y_1^{(\theta)}$ separates into a $\tau$-free part and a
$\tau$-linear part.  Define
\[
  f = f(k, \theta) = y_0/k + \theta\,y_1^{(\theta)},
  \qquad x = f + \tau\,y_1^{(0)}.
\]
Then
\[
  x^T Q x = f^T Q f + 2\tau\,f^T Q\,y_1^{(0)}
            + \tau^2\,(y_1^{(0)})^T Q\,y_1^{(0)},
\]
and the $1/\tau$ scaling gives three terms at distinct powers of~$\tau$:
\[
  \frac{x^T Q x}{\tau}
  = \underbrace{\frac{f^T Q f}{\tau}}_{\tau^{-1}}
  + \underbrace{2\,f^T Q\,y_1^{(0)}}_{\tau^0}
  + \underbrace{\tau\,(y_1^{(0)})^T Q\,y_1^{(0)}}_{\tau^1}.
\]

\subsection{Duality residual}

The absolute error in the identity is
\[
  r_{\mathrm{dual}} = \Bigl|b^T \lambda + c^T x + d^T\nu
    + \frac{x^T Q x}{\tau} + \frac{\mu}{\tau}
    - \theta\,(\langle b,e\rangle + 1)\Bigr|.
\]

\section{Joint $(k,\tau)$ selection with $\theta = \mu = 1/k^2$}

Setting $\theta = \mu = 1/k^2$ ties the centering weight to the barrier
parameter.  The direction becomes
\[
  d = d_0 + k\tau\,d_1^{(0)} + \frac{1}{k}\,d_1^{(\theta)},
\]
which depends on two free parameters $(k, \tau)$.

\subsection{Inner-product basis}

All norms and inner products of $d$ can be expressed in terms of
six scalars computed once from the decomposition:
\begin{alignat*}{3}
  &a = \|d_0\|^2, &\qquad
  &f = \langle d_0,\, d_1^{(0)}\rangle, &\qquad
  &g = \langle d_0,\, d_1^{(\theta)}\rangle, \\
  &p = \|d_1^{(0)}\|^2, &\qquad
  &q = \langle d_1^{(0)},\, d_1^{(\theta)}\rangle, &\qquad
  &r = \|d_1^{(\theta)}\|^2.
\end{alignat*}

\subsection{Optimal $\tau$ at fixed $k$}

The squared norm
\[
  \|d\|^2 = a + 2k\tau f + \frac{2g}{k}
            + k^2\tau^2 p + 2\tau q + \frac{r}{k^2}
\]
is quadratic in~$\tau$.  Setting $\partial\|d\|^2/\partial\tau = 0$:
\[
  \boxed{\tau^*(k) = -\frac{k\,f + q}{k^2\,p}.}
\]

\subsection{Reduced objective in $k$ alone}

Substituting $\tau^*(k)$ back:
\[
  \|d\|^2\bigl(k,\,\tau^*(k)\bigr)
  = a + \frac{2g}{k} + \frac{r}{k^2}
    - \frac{(k\,f + q)^2}{k^2\,p}.
\]
This is a rational function of~$k$ involving only the six precomputed
scalars.  It is minimized by a one-dimensional search over $k > 0$.

\subsection{Weighted $\tau$ selection with duality constraint}

The min-norm $\tau^*(k)$ above ignores the duality residual.  To
enforce the identity from Section~6, we select $\tau$ to minimize a
weighted objective that penalizes both $\|d\|^2$ and the squared
violation.

For fixed $(k,\theta)$, expand the duality identity.  The cone and
linear-cost contributions are affine in~$\tau$; the quadratic cost adds
a $\tau^{-1}$, $\tau^0$, and $\tau^1$ term (Section~6.2):
\[
  \underbrace{(\sigma_0 + \gamma_0 + 2\,f^T Q y_1^{(0)})}_{\alpha}
  + \underbrace{\bigl(\sigma_1 + \gamma_1
    + (y_1^{(0)})^T Q\,y_1^{(0)}\bigr)}_\beta \,\tau
  + \frac{\mu + f^T Q f}{\tau}
  = \underbrace{\theta\,(\langle b,e\rangle + 1)}_{R},
\]
where
\begin{align*}
  \sigma_0 &= \tfrac{1}{k}\,\langle b,\, \PW(e + d_0 + k\theta\,d_1^{(\theta)})\rangle,
  &\quad \sigma_1 &= \langle b,\, \PW(d_1^{(0)})\rangle, \\
  \gamma_0 &= c^T y_0 / k + \theta\,c^T y_1^{(\theta)},
  &\quad \gamma_1 &= c^T y_1^{(0)},
\end{align*}
and $f = y_0/k + \theta\,y_1^{(\theta)}$ is the $\tau$-free part of~$x$.
Multiplying through by~$\tau$ gives the \emph{violation quadratic}:
\[
  V(\tau) = \beta\,\tau^2 + (\alpha - R)\,\tau + \mu_{\mathrm{eff}},
\]
with
\begin{align*}
  \beta &= \sigma_1 + \gamma_1 + (y_1^{(0)})^T Q\,y_1^{(0)}, \\
  \alpha &= \sigma_0 + \gamma_0 + 2\,f^T Q\,y_1^{(0)}, \\
  \mu_{\mathrm{eff}} &= \mu + f^T Q f.
\end{align*}
When $Q = 0$ these reduce to the LP case.  The three Q inner products
$(y_1^{(0)})^T Q\,y_1^{(0)}$, $f^T Q\,y_1^{(0)}$, $f^T Q f$ are
computed via \texttt{AccumulateQx}; only the first is
$\tau$-independent.

The $\|d\|^2$ terms, grouping the $\tau$-independent part as
$h = d_0 + k\theta\,d_1^{(\theta)}$, give
\[
  \|d\|^2 = \|h\|^2 + 2k\,\langle h, d_1^{(0)}\rangle\,\tau
            + k^2 p\,\tau^2.
\]

The weighted objective is
\[
  \boxed{J(\tau) = \|d\|^2 + w\,V(\tau)^2,}
\]
where $w > 0$ is a penalty weight.  For $w \to \infty$ this enforces
$V(\tau) = 0$ exactly (the duality identity); for $w = 0$ it recovers
the pure min-norm selection.

The derivative $J'(\tau)$ is cubic in~$\tau$, so the minimizer can be
found by golden-section search over $\tau > 0$.

\subsection{Hard constraint ($w \to \infty$)}

In the limit $w \to \infty$, $\tau$ is determined by the roots of the
violation quadratic $V(\tau) = 0$:
\[
  \boxed{\tau = \frac{(R - \alpha) \pm \sqrt{(R-\alpha)^2 - 4\beta\,\mu_{\mathrm{eff}}}}
                     {2\beta}.}
\]
When both roots are positive, choose the one with smaller~$\|d\|^2$.
When only one root is positive, use it.  This enforces the duality
identity exactly and eliminates the tuning parameter~$w$.



\subsection{Algorithm (per outer iteration)}

\begin{enumerate}
  \item Compute the decomposition $(d_0,\, d_1^{(0)},\, d_1^{(\theta)})$:
        one factorization and three back-solves.
  \item Compute the six inner products $a, f, g, p, q, r$.
  \item Compute the duality coefficients $\sigma_1, \gamma_1$
        (once per factorization) and $\sigma_0, \gamma_0$
        (depend on $k$ via $\theta$).
  \item For the current $k$ (with $\theta = 1/k^2$), minimize
        $J(\tau)$ over $\tau > 0$.
  \item Evaluate $d = d_0 + k\tau\,d_1^{(0)} + (1/k)\,d_1^{(\theta)}$.
  \item Take the geodesic step $W \leftarrow W \cdot \exp(\alpha\,d)$
        with $\alpha = \min(1,\; 2/\|d\|_\infty^2)$.
\end{enumerate}

Centering terminates when both $\|d\|_\infty$ and the duality residual
$|V(\tau)/\tau|$ fall below a threshold.  The outer loop decreases
$\theta = \mu = 1/k^2$ (e.g.\ $\theta \leftarrow 0.9\,\theta$) and
re-centers at each level until $\theta < \varepsilon$.
